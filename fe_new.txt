  const handleSubmit = async (e) => {
    e.preventDefault();
    setSaving(true);
    // Send the whole profile + float_riders. The server treats the member's
    // own float number as authoritative for membership, but only when the client
    // explicitly provides it; otherwise it preserves the existing link.
    const body = {
      ...profile,
      float_riders: riders,
    };
    try {
      const res = await fetch('/api/users/me', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
      });
      if (!res.ok) {
        const err = await res.json().catch(() => ({}));
        throw new Error(err.error || 'Failed to save profile');
      }
      const data = await res.json();
      if (data.profile) setProfile(data.profile);
      setMsg('Profile saved');
    } catch (err) {
      setMsg(err.message || 'Error saving profile');
    } finally {
      setSaving(false);
    }
  };
