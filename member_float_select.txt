async function get__api_floats(req, res) {
  try {
    const result = await pool.query(
      `SELECT id, name, float_number
       FROM floats
       ORDER BY float_number NULLS LAST, name`,
    );
    res.json(result.rows.map((r) => ({ id: r.id, name: r.name, float_number: r.float_number })));
  } catch (error) {
    console.error('Failed to fetch floats', error);
    res.status(500).json({ error: 'Unable to fetch floats' });
  }
}

  set('pd-float-number', profile.member_float_number);

  // Constrain float selection to floats defined by the float admin (plus "None").
  // Members may not type an arbitrary float number.
  (async () => {
    const sel = document.getElementById('pd-float-number');
    if (!sel) return;
    try {
      const res = await fetch('/api/floats', { headers: { Authorization: 'Bearer ' + getToken() } });
      if (res.ok) {
        const floats = await res.json();
        const current = (profile.member_float_number || '').toString().trim();
        sel.innerHTML = '<option value="">None</option>' + (Array.isArray(floats) ? floats : [])
          .filter((f) => f && (f.float_number || '').toString().trim() !== '')
          .map((f) => {
            const num = (f.float_number || '').toString().trim();
            const label = `Float #${escHtml(num)} – ${escHtml(f.name || 'Unnamed')}`;
            return `<option value="${escHtml(num)}"${num === current ? ' selected' : ''}>${label}</option>`;
          }).join('');
        sel.value = current;
      }
    } catch { /* keep default "None" if the list can't be loaded */ }
  })();
